# Optimized bundle script with better performance and error handling
param(
    [string]$OutputFile = "all_scripts_bundle.txt",
    [string[]]$ExcludeFolders = @("build", "bin", ".git", "obj", "dist", "__pycache__")
)

# Performance optimization: Use .NET methods for file operations
$OutputPath = [System.IO.Path]::GetFullPath($OutputFile)
$Extensions = @("*.h", "*.cpp", "*.ino", "*.py", "*.sh", "*.ps1")

Write-Host "Bundling scripts into $OutputPath..." -ForegroundColor Cyan

# Clear output file efficiently
[System.IO.File]::WriteAllText($OutputPath, "", [System.Text.Encoding]::UTF8)

# Get files with parallel processing for better performance
$Files = Get-ChildItem -Recurse -File -Include $Extensions | Where-Object { 
    $filePath = $_.FullName
    $ExcludeFolders | ForEach-Object { $filePath -notmatch $_ }
}

# Process files with progress tracking
$counter = 0
$total = $Files.Count

$Files | ForEach-Object {
    $File = $_
    $counter++
    $percentComplete = ($counter / $total) * 100
    
    Write-Progress -Activity "Bundling Files" -Status "Processing: $($File.Name)" -PercentComplete $percentComplete
    Write-Host "Processing ($counter/$total): $($File.Name)" -ForegroundColor Yellow
    
    try {
        # Read file content efficiently
        $content = [System.IO.File]::ReadAllText($File.FullName, [System.Text.Encoding]::UTF8)
        
        # Create header with relative path
        $relativePath = $File.FullName.Replace((Get-Location).Path + "\", "")
        $Header = "`n`n--- FILE: $relativePath ---`n`n"
        
        # Append to file using efficient method
        [System.IO.File]::AppendAllText($OutputPath, $Header + $content, [System.Text.Encoding]::UTF8)
    }
    catch {
        Write-Host "Error processing $($File.Name): $($_.Exception.Message)" -ForegroundColor Red
    }
}

Write-Progress -Activity "Bundling Files" -Completed
Write-Host "Done! Bundled $total files into $OutputPath" -ForegroundColor Green

# Show file information
$fileInfo = Get-Item $OutputPath
Write-Host "Output file size: $([math]::Round($fileInfo.Length/1KB, 2)) KB" -ForegroundColor Cyan

#include <Arduino.h>
#include <Wire.h>
#include "config.h"
#include "miner_core.h"
#include "i2c_slave.h"

#ifdef LCD
#include "display_manager.h"
DisplayManager display;
#endif

MinerCore miner;
I2CSlave* i2c_slave = nullptr;

uint8_t detectAddressFromLadder() {
    int adc_val = analogRead(LADDER_PIN);
    uint8_t slot = adc_val / 128; 
    uint8_t finalAddr = I2C_BASE_ADDRESS + slot;
    Serial.printf("[Hardware] Pin: %d | ADC: %d | Addr: 0x%02X\n", LADDER_PIN, adc_val, finalAddr);
    return finalAddr;
}

void setup() {
    Serial.begin(115200);
    delay(500);
    
    // Disable Watchdog on Core 0 to prevent reset during long hashing tasks
    disableCore0WDT();

    uint8_t myAddr = detectAddressFromLadder();

    #ifdef LCD
    display.begin();
    display.showBootScreen();
    delay(1000); // Give time for the user to see the address
    #endif

    miner.begin();

    i2c_slave = new I2CSlave(myAddr);
    i2c_slave->begin(myAddr);

    pinMode(LED_PIN, OUTPUT);
    
    #ifdef LCD
    display.showMiningScreen(); // Transitions from Boot to Mining layout
    #endif
}

void loop() {
    // 1. I2C Job Handling
    if (i2c_slave->m_new_job_available) {
        i2c_slave->m_new_job_available = false;
        
        JobRequest m_job;
        m_job.job_id = i2c_slave->m_current_job.id;
        m_job.difficulty = i2c_slave->m_current_job.difficulty;
        m_job.nonce_start = i2c_slave->m_current_job.nonce_start;
        m_job.nonce_range = NONCES_PER_JOB;
        memcpy(m_job.header_bytes, i2c_slave->m_current_job.header, 76);
        
        miner.setNewJob(m_job);
        digitalWrite(LED_PIN, HIGH);
    }

    // 2. The Hashing Core
    miner.run(); 

    // 3. Stats Aggregation
    uint32_t hashes = miner.getAndResetHashes();
    if (hashes > 0) {
        i2c_slave->m_hashes_since_poll += hashes;
        digitalWrite(LED_PIN, LOW);
    }

    uint32_t found = miner.getFoundNonce();
    if (found != 0xFFFFFFFF) {
        i2c_slave->m_found_nonce = found;
    }

    // 4. Display Update Logic
    #ifdef LCD
    display.handleButtons();
    
    static uint32_t last_display_update = 0;
    uint32_t now = millis();
    if (now - last_display_update >= DISPLAY_UPDATE_INTERVAL) {
        DisplayStats stats;
        // Map data to the struct used by display_manager
        stats.hashrate = (double)hashes / (DISPLAY_UPDATE_INTERVAL / 1000.0);
        stats.total_hashes = i2c_slave->m_hashes_since_poll;
        stats.mining_active = (hashes > 0);
        stats.shares_found = (found != 0xFFFFFFFF) ? 1 : 0; 
        stats.jobs_received = 1; // Counter can be expanded in miner_core
        stats.crc_errors = 0;
        stats.core0_active = true;
        stats.core1_active = true;
        
        display.updateStats(stats);
        last_display_update = now;
    }
    #endif
}

I2CSlave* I2CSlave::s_instance = nullptr;

I2CSlave::I2CSlave(uint8_t addr) : m_addr(addr), m_new_job_available(false) {
    s_instance = this;
    m_hashes_since_poll.store(0);
    m_found_nonce.store(0xFFFFFFFF);
    m_last_heartbeat = millis();
}

void I2CSlave::begin(uint8_t address) {
    Wire.begin(address, I2C_SDA_PIN, I2C_SCL_PIN, I2C_CLOCK_SPEED);
    Wire.onReceive(handleReceive);
    Wire.onRequest(handleRequest);
}

void I2CSlave::handleReceive(int len) {
    if (!s_instance) return;

    // Small command (Ping/Reset/Stop)
    if (len == 1) {
        uint8_t cmd = Wire.read();
        if (cmd == MINER_CMD_PING) {
            s_instance->m_last_heartbeat = millis();
        } else if (cmd == MINER_CMD_RESET) {
            ESP.restart();
        }
        return;
    }

    // Large packet (Job Feed)
    if (len == sizeof(JobI2cRequest)) {
        JobI2cRequest temp;
        Wire.readBytes((uint8_t*)&temp, sizeof(JobI2cRequest));
        
        // Verify CRC before accepting job
        if (s_instance->verifyCRC(&temp, sizeof(temp))) {
            s_instance->m_current_job = temp;
            s_instance->m_new_job_available = true;
            s_instance->m_last_heartbeat = millis();
        }
    }
}

void I2CSlave::handleRequest() {
    if (!s_instance) return;

    JobI2cResult res;
    res.cmd = MINER_CMD_SLAVE_RESULT;
    res.id = s_instance->m_current_job.id;
    res.reserved = 0;
    res.nonce = s_instance->m_found_nonce.load();
    res.processed_nonce = s_instance->m_hashes_since_poll.load();

    // Reset stats
    s_instance->m_hashes_since_poll.store(0);
    s_instance->m_found_nonce.store(0xFFFFFFFF);

    // CRITICAL: Calculate CRC over the first 11 bytes, store in 12th byte
    // Or calculate over all bytes with CRC field as 0
    res.crc = 0;
    res.crc = s_instance->calculateCRC8(&res, sizeof(res));
    
    // Send exactly 12 bytes
    Wire.write((uint8_t*)&res, sizeof(JobI2cResult));
}

uint8_t I2CSlave::calculateCRC8(const void* data, size_t len) {
    uint8_t crc = 0xff;
    const uint8_t* p = (const uint8_t*)data;
    for (size_t i = 0; i < len; i++) {
        crc ^= p[i];
        for (int j = 0; j < 8; j++) {
            if (crc & 0x80) crc = (crc << 1) ^ 0x31;
            else crc <<= 1;
        }
    }
    return crc;
}

bool I2CSlave::verifyCRC(void* data, size_t len) {
    uint8_t* p = (uint8_t*)data;
    uint8_t received_crc = p[1];
    p[1] = 0;
    uint8_t calc = calculateCRC8(data, len);
    p[1] = received_crc;
    return calc == received_crc;
}

    digitalWrite(LED_PIN, on ? HIGH : LOW);
    m_led_state = on;
}

void LedManager::setPattern(LedPattern pattern) {
    m_current_pattern.store(pattern);
    m_pattern_state = 0;
    m_last_update = millis();
}

void LedManager::update() {
    uint32_t now = millis();
    updatePattern(m_current_pattern.load(), now);
}

void LedManager::updatePattern(LedPattern pattern, uint32_t now) {
    // Minimum update interval (prevent flickering)
    if (now - m_last_update < 10) return;
    
    switch (pattern) {
        case LED_OFF:
            setLed(false);
            break;
            
        case LED_MINING_IDLE:
            // Slow pulse (2 seconds on, 2 seconds off)
            if (now - m_last_update >= 2000) {
                m_pattern_state = !m_pattern_state;
                setLed(m_pattern_state);
                m_last_update = now;
            }
            break;
            
        case LED_MINING_ACTIVE:
            // Fast pulse (500ms on, 500ms off)
            if (now - m_last_update >= 500) {
                m_pattern_state = !m_pattern_state;
                setLed(m_pattern_state);
                m_last_update = now;
            }
            break;
            
        case LED_SHARE_FOUND:
            // Bright flash (3 quick blinks)
            if (now - m_last_update >= 150) {
                m_pattern_state++;
                if (m_pattern_state >= 6) {  // 3 on/off cycles
                    setPattern(LED_MINING_ACTIVE);  // Return to active
                    m_pattern_state = 0;
                } else {
                    setLed(m_pattern_state % 2 == 0);
                }
                m_last_update = now;
            }
            break;
            
        case LED_ERROR:
            // Fast blink (200ms on, 200ms off)
            if (now - m_last_update >= 200) {
                m_pattern_state = !m_pattern_state;
                setLed(m_pattern_state);
                m_last_update = now;
            }
            break;
            
        case LED_NO_JOB:
            // Very slow blink (3 seconds on, 3 seconds off)
            if (now - m_last_update >= 3000) {
                m_pattern_state = !m_pattern_state;
                setLed(m_pattern_state);
                m_last_update = now;
            }
            break;
    }
}

#include "miner_core.h"
#include "mbedtls/sha256.h"
#include "config.h"

// Initialize shared variables
MinerCore::MinerCore() : m_found_nonce(0xFFFFFFFF), m_hashes_done(0), m_new_job(false) {}

void MinerCore::begin() {
    mbedtls_sha256_init(&m_ctx_static);
    mbedtls_sha256_init(&m_ctx_active);
}

void MinerCore::setNewJob(const JobRequest& job) {
    m_current_job = job;
    
    // PRE-COMPUTE MIDSTATE
    // We hash the first 64 bytes of the 80-byte header once per job.
    // This removes 80% of the hashing workload from the inner loop.
    mbedtls_sha256_starts_ret(&m_ctx_static, 0);
    mbedtls_sha256_update_ret(&m_ctx_static, m_current_job.header_bytes, 64);
    
    m_new_job = true; 
}

void MinerCore::run() {
    uint8_t hash_mid[32];
    uint8_t hash_final[32];
    uint8_t* header_ptr = (uint8_t*)m_current_job.header_bytes;

    while(true) {
        if (!m_new_job) {
            delay(10);
            continue;
        }
        
        m_new_job = false;
        uint32_t end_nonce = m_current_job.nonce_start + NONCES_PER_JOB;

        for (uint32_t nonce = m_current_job.nonce_start; nonce < end_nonce; nonce++) {
            // CLONE: Copy the pre-computed static state (Extremely Fast)
            mbedtls_sha256_clone(&m_ctx_active, &m_ctx_static);

            // UPDATE: Hash only the last 16 bytes (contains the changing nonce)
            *(uint32_t*)(header_ptr + 76) = nonce;
            mbedtls_sha256_update_ret(&m_ctx_active, header_ptr + 64, 16);
            mbedtls_sha256_finish_ret(&m_ctx_active, hash_mid);

            // SHA256d: Second pass
            mbedtls_sha256_starts_ret(&m_ctx_active, 0);
            mbedtls_sha256_update_ret(&m_ctx_active, hash_mid, 32);
            mbedtls_sha256_finish_ret(&m_ctx_active, hash_final);

            // Check difficulty (Fast zero-byte check)
            if (hash_final[31] == 0 && hash_final[30] == 0) {
                m_found_nonce = nonce;
                break; // Report share immediately
            }

            m_hashes_done++;
            
            // Periodically check if we need to yield for I2C interrupts
            if ((nonce & 0x3FF) == 0 && m_new_job) break; 
        }
    }
}

uint32_t MinerCore::getAndResetHashes() {
    return m_hashes_done.exchange(0);
}

IRAM_ATTR uint8_t crc8_compute(const void* data, size_t len) {
    const uint8_t* ptr = (const uint8_t*)data;
    uint8_t crc = 0;
    for (size_t i = 0; i < len; i++) {
        crc = CRC8_TABLE[crc ^ ptr[i]];
    }
    return crc;
}

// Optimized double SHA256 - mbedTLS on ESP32 auto-uses hardware SHA
IRAM_ATTR void sha256_double(const uint8_t* data, size_t len, uint8_t* hash) {
    uint8_t intermediate[32];
    mbedtls_sha256_context ctx;
    
    mbedtls_sha256_init(&ctx);
    mbedtls_sha256_starts_ret(&ctx, 0);
    mbedtls_sha256_update_ret(&ctx, data, len);
    mbedtls_sha256_finish_ret(&ctx, intermediate);
    
    mbedtls_sha256_starts_ret(&ctx, 0);
    mbedtls_sha256_update_ret(&ctx, intermediate, 32);
    mbedtls_sha256_finish_ret(&ctx, hash);
    mbedtls_sha256_free(&ctx);
}
